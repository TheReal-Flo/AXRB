cmake_minimum_required(VERSION 3.22)
project(axrb_windows_clock LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
    GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
add_library(axrb_whpx_clock SHARED clock_hook.cpp)
target_link_libraries(axrb_whpx_clock PRIVATE minhook)
add_executable(axrb_clock_launcher launcher.cpp)
target_link_libraries(axrb_clock_launcher PRIVATE psapi)
add_executable(axrb_clock_filter_test filter_test.cpp)
foreach(target axrb_whpx_clock axrb_clock_launcher axrb_clock_filter_test)
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
endforeach()
enable_testing()
add_test(NAME clock_register_filter COMMAND axrb_clock_filter_test)
add_custom_command(TARGET axrb_whpx_clock POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy
    "${minhook_SOURCE_DIR}/LICENSE.txt" "$<TARGET_FILE_DIR:axrb_whpx_clock>/MinHook-LICENSE.txt")
