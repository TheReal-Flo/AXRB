cmake_minimum_required(VERSION 3.22)
project(axrb_windows_gpu_layer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(axrb_capture_shared capture_shared.cpp)
target_link_libraries(axrb_capture_shared PRIVATE d3d11 dxgi)
set(AXRB_VULKAN_HEADERS "" CACHE PATH "Optional Vulkan headers directory")
if(NOT AXRB_VULKAN_HEADERS)
  file(GLOB _ndk_includes "$ENV{LOCALAPPDATA}/Android/Sdk/ndk/*/toolchains/llvm/prebuilt/windows-x86_64/sysroot/usr/include")
  list(SORT _ndk_includes COMPARE NATURAL ORDER DESCENDING)
  if(NOT _ndk_includes)
    message(FATAL_ERROR "Install an Android NDK or set AXRB_VULKAN_HEADERS")
  endif()
  list(GET _ndk_includes 0 _ndk_include)
  # Copy only Khronos headers. Adding the entire Android sysroot would shadow
  # Windows CRT headers such as stdint.h.
  file(COPY "${_ndk_include}/vulkan" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/headers")
  if(EXISTS "${_ndk_include}/vk_video")
    file(COPY "${_ndk_include}/vk_video" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/headers")
  endif()
  set(AXRB_VULKAN_HEADERS "${CMAKE_CURRENT_BINARY_DIR}/headers")
endif()
add_executable(axrb_gpu_interop_probe interop_probe.cpp)
add_executable(axrb_descriptor_template_smoke ../../tests/vulkan_descriptor_template_smoke.cpp)
target_include_directories(axrb_descriptor_template_smoke PRIVATE "${AXRB_VULKAN_HEADERS}")
target_include_directories(axrb_gpu_interop_probe PRIVATE "${AXRB_VULKAN_HEADERS}")
target_compile_definitions(axrb_gpu_interop_probe PRIVATE VK_USE_PLATFORM_WIN32_KHR VK_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(axrb_gpu_interop_probe PRIVATE d3d11 dxgi)
add_library(axrb_gpu_layer SHARED layer.cpp layer.def)
set_property(TARGET axrb_gpu_layer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_include_directories(axrb_gpu_layer PRIVATE "${AXRB_VULKAN_HEADERS}" ../../protocol)
target_compile_definitions(axrb_gpu_layer PRIVATE VK_USE_PLATFORM_WIN32_KHR VK_NO_PROTOTYPES WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(axrb_gpu_layer PRIVATE d3d11 dxgi)
file(GENERATE OUTPUT "$<TARGET_FILE_DIR:axrb_gpu_layer>/axrb_gpu_layer.json" CONTENT "{\n\"file_format_version\":\"1.2.0\",\n\"layer\":{\"name\":\"VK_LAYER_AXRB_gpu_share\",\"type\":\"GLOBAL\",\"library_path\":\"$<TARGET_FILE:axrb_gpu_layer>\",\"api_version\":\"1.3.0\",\"implementation_version\":\"1\",\"description\":\"AXRB Windows shared eye textures\"}}\n")

add_executable(axrb_gpu_receiver_probe receiver_probe.cpp)
target_include_directories(axrb_gpu_receiver_probe PRIVATE ../../protocol)
target_compile_definitions(axrb_gpu_receiver_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(axrb_gpu_receiver_probe PRIVATE d3d11 dxgi)
